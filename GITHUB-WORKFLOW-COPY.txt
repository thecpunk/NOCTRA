name: Build NOCTRA APK from ZIP

on:
  workflow_dispatch:
  push:
    branches: [ main, master ]

permissions:
  contents: read

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Unpack NOCTRA project
        run: |
          test -f NOCTRA-Android-Project-Monochrome.zip
          mkdir source
          unzip -q NOCTRA-Android-Project-Monochrome.zip -d source
          test -f source/NOCTRA/app/build.gradle.kts

      - name: Set up JDK 17
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'

      - name: Set up Android SDK
        uses: android-actions/setup-android@v3

      - name: Install Android platform and build tools
        run: sdkmanager "platforms;android-35" "build-tools;35.0.0"

      - name: Set up Gradle
        uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: '8.9'

      - name: Build debug APK
        working-directory: source/NOCTRA
        run: gradle --no-daemon assembleDebug

      - name: Upload APK artifact
        uses: actions/upload-artifact@v4
        with:
          name: NOCTRA-debug-apk
          path: source/NOCTRA/app/build/outputs/apk/debug/*.apk
          if-no-files-found: error
          retention-days: 14
